# --- Local OCR (PaddleOCR) ---
# numpy is a light dependency needed just to decode images for OCR, so
# it's always installed. paddleocr/paddlepaddle are the actual OCR
# engine and are heavy (paddlepaddle alone is several hundred MB) —
# they're only ever imported lazily by app/services/paddle_ocr.py, so
# a plain `pip install -r requirements.txt` (including in CI, where
# the real engine is mocked out in tests and never instantiated) still
# works, it's just a bigger download.
numpy>=1.26,<2.0
paddlepaddle==3.0.0
paddleocr==3.5.0
# PaddleOCR's own image decoding uses opencv; the headless build avoids
# pulling in GUI/X11 libraries this server will never use.
